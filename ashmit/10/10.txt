import pandas as pd

col_names = [
    "SepalLengthCm",
    "SepalWidthCm",
    "PetalLengthCm",
    "PetalWidthCm",
    "Species"
]

data = pd.read_csv("Iris.csv", names=col_names)

data.head()

# g. Create data subsets for different species

setosa = data[data["Species"] == "Iris-setosa"]

versicolor = data[data["Species"] == "Iris-versicolor"]

virginica = data[data["Species"] == "Iris-virginica"]

setosa.head()

versicolor.head()

virginica.head()

# h. Merge two subsets

merged_data = pd.concat([setosa, versicolor])

merged_data.head()

# i. Sort Data using Petal Length

sorted_data = merged_data.sort_values(
    by="PetalLengthCm",
    ascending=False
)

sorted_data.head()

# j. Transposing Data

transposed_data = sorted_data.T

transposed_data.head()

# k. Melting Data to long format

melted_data = pd.melt(
    data,
    id_vars=["Species"],
    value_vars=[
        "SepalLengthCm",
        "SepalWidthCm",
        "PetalLengthCm",
        "PetalWidthCm"
    ]
)

melted_data.head()

# l. Casting data to wide format

casted_data = melted_data.pivot_table(
    index="Species",
    columns="variable",
    values="value",
    aggfunc="mean"
)

casted_data.head()
