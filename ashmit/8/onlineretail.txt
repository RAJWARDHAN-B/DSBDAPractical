hive

create table online_retail(
id int,
customer_name string,
country string,
product string,
quantity int,
price int,
total int)
row format delimited
fields terminated by ',';

load data local inpath '/home/cloudera/online_retail.txt'
into table online_retail;

select * from online_retail;

create index retail_index
on table online_retail(product)
as 'org.apache.hadoop.hive.ql.index.compact.CompactIndexHandler'
with deferred rebuild;

show index on online_retail;

select sum(total) as total_sales
from online_retail;

select avg(total) as average_sales
from online_retail;

select *
from online_retail
order by total desc
limit 1;

select customer_name,sum(total) as order_total
from online_retail
group by customer_name
order by order_total desc
limit 1;

select country,sum(total) as total_sales
from online_retail
group by country
order by total_sales desc
limit 1;

select country,sum(total) as total_sales
from online_retail
group by country
order by total_sales asc
limit 1;

create external table hbase_retail(
id int,
customer_name string,
country string,
product string,
quantity int,
price int,
total int)
stored by 'org.apache.hadoop.hive.hbase.HBaseStorageHandler'
with serdeproperties(
"hbase.columns.mapping"=":key,rinfo:customer_name,rinfo:country,sales:product,sales:quantity,sales:price,sales:total")
tblproperties("hbase.table.name"="retail");

select * from hbase_retail;
